    db = getFirestore(app);

  console.warn("Firebase config not present, running in resilient in-memory mode.", err);

const formatZMW = (amount) => {
  return 'K' + Math.round(Number(amount || 0)).toLocaleString('en-US');
};

export default function App() {
  // Navigation & Auth State
  const [activeTab, setActiveTab] = useState('marketplace'); // 'marketplace' | 'sell' | 'admin'
  const [isAdminUnlocked, setIsAdminUnlocked] = useState(false);
  const [pinInput, setPinInput] = useState('');
  const [pinError, setPinError] = useState('');
  const [user, setUser] = useState(null);

  // Core Platform State
  const [commissionRate, setCommissionRate] = useState(5); // Default 5% middleman fee
  const [vehicles, setVehicles] = useState(INITIAL_VEHICLES);
  const [transactions, setTransactions] = useState(INITIAL_TRANSACTIONS);

  // Marketplace Filter State
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedMake, setSelectedMake] = useState('All');
  const [selectedFuel, setSelectedFuel] = useState('All');
  const [maxPrice, setMaxPrice] = useState(700000);

  // Modals & Notifications
  const [selectedVehicle, setSelectedVehicle] = useState(null);
  const [checkoutVehicle, setCheckoutVehicle] = useState(null);
  const [toast, setToast] = useState(null);

  // Seller Form State
  const [sellForm, setSellForm] = useState({
    make: 'Toyota',
    model: '',
    year: 2017,
    mileage: '',
    transmission: 'Automatic',
    fuel: 'Petrol',
    location: 'Lusaka',
    sellerPrice: '',
    sellerName: '',
    sellerPhone: '',
    description: '',
    image: 'https://images.unsplash.com/photo-1541899481282-d53bffe3c35d?auto=format&fit=crop&w=800&q=80'
  });

  // Checkout Modal State
  const [checkoutState, setCheckoutState] = useState({
    buyerName: '',
    buyerPhone: '',
    paymentOption: 'full', // 'full' | 'deposit' (10% escrow reservation)
    paymentMethod: 'momo', // 'momo' | 'bank' | 'card'
    momoNetwork: 'Airtel Money',
    momoNumber: '',
    bankName: 'Zanaco',
    bankRef: '',
    cardNumber: '',
    cardExpiry: '',
    cardCvv: '',
    isProcessing: false,
    receipt: null
  });

  // Admin Ledger Filter
  const [ledgerFilter, setLedgerFilter] = useState('All');

  const showToast = (message, type = 'success') => {
    setToast({ message, type });
    setTimeout(() => setToast(null), 4000);
  };

  // --- FIREBASE AUTH & REAL-TIME SYNC ---
  useEffect(() => {
    if (!auth) return;
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (e) {
        console.warn("Auth fallback:", e);
      }
    };
    initAuth();
    const unsub = onAuthStateChanged(auth, setUser);
    return () => unsub();
  }, []);

  useEffect(() => {
    if (!db || !user) return;
    const vehCol = collection(db, 'artifacts', appId, 'public', 'data', 'vehicles');
    const txnCol = collection(db, 'artifacts', appId, 'public', 'data', 'transactions');
    const cfgDoc = doc(db, 'artifacts', appId, 'public', 'data', 'settings', 'brokerConfig');

    const unsubVeh = onSnapshot(vehCol, (snap) => {
      if (!snap.empty) {
        const list = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        setVehicles(list);
      }
    }, (err) => console.warn("Vehicles sync error:", err));

    const unsubTxn = onSnapshot(txnCol, (snap) => {
      if (!snap.empty) {
        const list = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        setTransactions(list);
      }
    }, (err) => console.warn("Transactions sync error:", err));

    const unsubCfg = onSnapshot(cfgDoc, (snap) => {
      if (snap.exists() && snap.data().commissionRate) {
        setCommissionRate(snap.data().commissionRate);
      }
    }, (err) => console.warn("Config sync error:", err));

    return () => {
      unsubVeh();
      unsubTxn();
      unsubCfg();
    };
  }, [user]);

  // --- HELPER: PERSIST TO FIRESTORE IF AVAILABLE ---
  const syncVehicleToCloud = async (vehicleObj) => {
    if (!db || !user) return;
    try {
      await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'vehicles', vehicleObj.id), vehicleObj);
    } catch (e) {
      console.warn("Cloud vehicle write skipped:", e);
    }
  };

  const syncTransactionToCloud = async (txnObj) => {
    if (!db || !user) return;
    try {
      await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'transactions', txnObj.id), txnObj);
    } catch (e) {
      console.warn("Cloud txn write skipped:", e);
    }
  };

  const handleCommissionChange = async (newRate) => {
    const rate = Number(newRate);
    setCommissionRate(rate);
    if (db && user) {
      try {
        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'settings', 'brokerConfig'), { commissionRate: rate }, { merge: true });
      } catch (e) {
        console.warn("Cloud config write skipped:", e);
      }
    }
  };

  // --- DERIVED METRICS & FILTERED LISTINGS ---
  const approvedVehicles = useMemo(() => {
    return vehicles.filter(v => {
      if (v.status !== 'approved' && v.status !== 'escrow') return false;
      const totalWithFee = v.sellerPrice * (1 + commissionRate / 100);
      if (totalWithFee > maxPrice) return false;
      if (selectedMake !== 'All' && v.make !== selectedMake) return false;
      if (selectedFuel !== 'All' && v.fuel !== selectedFuel) return false;
      if (searchQuery.trim() !== '') {
        const q = searchQuery.toLowerCase();
        const match = v.title.toLowerCase().includes(q) || 
                      v.location.toLowerCase().includes(q) || 
                      v.make.toLowerCase().includes(q);
        if (!match) return false;
      }
      return true;
    });
  }, [vehicles, commissionRate, maxPrice, selectedMake, selectedFuel, searchQuery]);

  const pendingVehicles = useMemo(() => vehicles.filter(v => v.status === 'pending'), [vehicles]);

  const adminStats = useMemo(() => {
    const grossSales = transactions.reduce((sum, t) => sum + Number(t.totalPaid || 0), 0);
    const netCommissions = transactions.reduce((sum, t) => sum + Number(t.commissionAmount || 0), 0);
    const activeCount = vehicles.filter(v => v.status === 'approved').length;
    return {
      grossSales,
      netCommissions,
      activeCount,
      pendingCount: pendingVehicles.length
    };
  }, [transactions, vehicles, pendingVehicles]);

  // --- HANDLERS ---
  const handleAdminUnlock = (e) => {
    e.preventDefault();
    if (pinInput === '1234') {
      setIsAdminUnlocked(true);
      setPinError('');
      showToast('Broker Admin Dashboard unlocked!');
    } else {
      setPinError('Incorrect PIN. Use demo PIN: 1234');
    }
  };

  const handleSellerSubmit = async (e) => {
    e.preventDefault();
    if (!sellForm.model || !sellForm.sellerPrice || !sellForm.sellerName || !sellForm.sellerPhone) {
      showToast('Please fill in all required vehicle and contact fields.', 'error');
      return;
    }

    const newVeh = {
      id: 'veh-' + Math.floor(1000 + Math.random() * 9000),
      title: `${sellForm.year} ${sellForm.make} ${sellForm.model}`,
      make: sellForm.make,
      model: sellForm.model,
      year: Number(sellForm.year),
      mileage: sellForm.mileage || '75,000 km',
      transmission: sellForm.transmission,
      fuel: sellForm.fuel,
      location: sellForm.location || 'Lusaka',
      sellerPrice: Number(sellForm.sellerPrice),
      status: 'pending',
      sellerName: sellForm.sellerName,
      sellerPhone: sellForm.sellerPhone,
      description: sellForm.description || 'Verified local seller vehicle submitted via ZedAuto Broker.',
      image: sellForm.image || 'https://images.unsplash.com/photo-1541899481282-d53bffe3c35d?auto=format&fit=crop&w=800&q=80',
      createdAt: new Date().toISOString()
    };

    setVehicles(prev => [newVeh, ...prev]);
    await syncVehicleToCloud(newVeh);

    setSellForm({
      make: 'Toyota',
      model: '',
      year: 2017,
      mileage: '',
      transmission: 'Automatic',
      fuel: 'Petrol',
      location: 'Lusaka',
      sellerPrice: '',
      sellerName: '',
      sellerPhone: '',
      description: '',
      image: 'https://images.unsplash.com/photo-1541899481282-d53bffe3c35d?auto=format&fit=crop&w=800&q=80'
    });

    showToast('Vehicle submitted! Sent to Broker Moderation Queue for approval.');
    setActiveTab('marketplace');
  };

  const handleModerationAction = async (vehId, action) => {
    const updatedList = vehicles.map(v => {
      if (v.id === vehId) {
        return { ...v, status: action === 'approve' ? 'approved' : 'rejected' };
      }
      return v;
    });
    setVehicles(updatedList);
    const target = updatedList.find(v => v.id === vehId);
    if (target) await syncVehicleToCloud(target);
    showToast(action === 'approve' ? 'Vehicle approved and published live!' : 'Vehicle listing rejected.');
  };

  const openCheckout = (vehicle) => {
    setSelectedVehicle(null);
    setCheckoutVehicle(vehicle);
    setCheckoutState({
      buyerName: '',
      buyerPhone: '',
      paymentOption: 'full',
      paymentMethod: 'momo',
      momoNetwork: 'Airtel Money',
      momoNumber: '',
      bankName: 'Zanaco',
      bankRef: '',
      cardNumber: '',
      cardExpiry: '',
      cardCvv: '',
      isProcessing: false,
      receipt: null
    });
  };

  const handleCompletePayment = async (e) => {
    e.preventDefault();
    if (!checkoutState.buyerName || !checkoutState.buyerPhone) {
      showToast('Please enter your full name and phone number.', 'error');
      return;
    }

    setCheckoutState(prev => ({ ...prev, isProcessing: true }));

    // Simulate payment gateway / USSD push verification delay
    setTimeout(async () => {
      const sellerAmt = checkoutVehicle.sellerPrice;
      const fullCommission = Math.round(sellerAmt * (commissionRate / 100));
      const fullTotal = sellerAmt + fullCommission;

      const isDeposit = checkoutState.paymentOption === 'deposit';
      const amountPaidNow = isDeposit ? Math.round(fullTotal * 0.10) : fullTotal;
      const commissionEarnedNow = isDeposit ? Math.round(fullCommission * 0.10) : fullCommission;

      let methodLabel = '';
      let refCode = '';

      if (checkoutState.paymentMethod === 'momo') {
        methodLabel = `Mobile Money (${checkoutState.momoNetwork})`;
        refCode = `MOMO-${Math.floor(100000 + Math.random() * 900000)}`;
      } else if (checkoutState.paymentMethod === 'bank') {
        methodLabel = `Bank Transfer (${checkoutState.bankName})`;
        refCode = checkoutState.bankRef || `EFT-${Math.floor(100000 + Math.random() * 900000)}`;
      } else {
        methodLabel = 'Card Payment (Visa/Mastercard)';
        refCode = `CARD-${Math.floor(100000 + Math.random() * 900000)}`;
      }

      const newTxn = {
        id: 'TXN-' + Math.floor(1000 + Math.random() * 9000),
        vehicleId: checkoutVehicle.id,
        vehicleTitle: checkoutVehicle.title,
        buyerName: checkoutState.buyerName,
        buyerPhone: checkoutState.buyerPhone,
        sellerName: checkoutVehicle.sellerName,
        sellerAmount: isDeposit ? Math.round(sellerAmt * 0.10) : sellerAmt,
        commissionAmount: commissionEarnedNow,
        totalPaid: amountPaidNow,
        paymentType: isDeposit ? '10% Escrow Deposit' : 'Full Purchase',
        paymentMethod: methodLabel,
        reference: refCode,
        escrowStatus: 'Held in Broker Escrow',
        date: new Date().toLocaleString('en-GB', { dateStyle: 'short', timeStyle: 'short' })
      };

      // Update vehicle status to escrow/sold
      const updatedVehicles = vehicles.map(v =>
        v.id === checkoutVehicle.id ? { ...v, status: isDeposit ? 'escrow' : 'sold' } : v
      );
      setVehicles(updatedVehicles);
      const updatedVeh = updatedVehicles.find(v => v.id === checkoutVehicle.id);
      if (updatedVeh) await syncVehicleToCloud(updatedVeh);

      setTransactions(prev => [newTxn, ...prev]);
      await syncTransactionToCloud(newTxn);

      setCheckoutState(prev => ({
        ...prev,
        isProcessing: false,
        receipt: newTxn
      }));

      showToast(`Payment confirmed! ${formatZMW(commissionEarnedNow)} commission credited to Broker Wallet.`);
    }, 1200);
  };

  const handleReleaseEscrow = async (txnId) => {
    const updatedTxns = transactions.map(t =>
      t.id === txnId ? { ...t, escrowStatus: 'Released to Seller' } : t
    );
    setTransactions(updatedTxns);
    const target = updatedTxns.find(t => t.id === txnId);
    if (target) await syncTransactionToCloud(target);
    showToast('Seller funds marked as released from escrow!');
  };

  const uniqueMakes = ['All', ...Array.from(new Set(vehicles.map(v => v.make)))];

  return (
    <div className="min-h-screen bg-slate-50 text-slate-900 font-sans pb-16">
      {/* Toast Notification Banner */}
      {toast && (
        <div className={`fixed bottom-5 right-5 z-50 flex items-center gap-3 px-5 py-3.5 rounded-xl shadow-xl border text-sm font-medium transition-all animate-bounce ${
          toast.type === 'error' 
            ? 'bg-red-900 text-white border-red-700' 
            : 'bg-slate-900 text-white border-emerald-500/40'
        }`}>
          {toast.type === 'error' ? <AlertCircle className="w-5 h-5 text-red-400" /> : <CheckCircle2 className="w-5 h-5 text-emerald-400" />}
          <span>{toast.message}</span>
        </div>
      )}

      {/* Top Navigation Header */}
      <header className="sticky top-0 z-40 bg-slate-900 text-white border-b border-slate-800 shadow-md">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
          <div className="flex flex-col md:flex-row md:items-center md:justify-between py-3.5 gap-3">
            {/* Brand Logo */}
            <div className="flex items-center justify-between">
              <div 
                onClick={() => setActiveTab('marketplace')}
                className="flex items-center gap-2.5 cursor-pointer group"
              >
                <div className="bg-emerald-500 p-2 rounded-xl text-slate-950 shadow-lg shadow-emerald-500/20 group-hover:scale-105 transition">
                  <Car className="w-6 h-6 stroke-[2.5]" />
                </div>
                <div>
                  <div className="flex items-center gap-2">
                    <span className="text-xl font-extrabold tracking-tight">ZedAuto<span className="text-emerald-400">Broker</span></span>
                    <span className="text-[10px] uppercase tracking-wider font-bold bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded-full border border-emerald-500/30">
                      Lusaka Escrow
                    </span>
                  </div>
                  <p className="text-xs text-slate-400 hidden sm:block">Verified Vehicle Marketplace & Secure Middleman</p>
                </div>
              </div>

              {/* Live Commission Wallet Pill (Mobile visible) */}
              <div 
                onClick={() => setActiveTab('admin')}
                className="flex md:hidden items-center gap-1.5 bg-slate-800 hover:bg-slate-700 border border-slate-700 px-3 py-1.5 rounded-lg cursor-pointer text-xs"
              >
                <Wallet className="w-3.5 h-3.5 text-emerald-400" />
                <span className="font-bold text-emerald-400">{formatZMW(adminStats.netCommissions)}</span>
              </div>
            </div>

            {/* Navigation Switcher */}
            <nav className="flex items-center gap-1.5 bg-slate-800/90 p-1 rounded-xl border border-slate-700/80 overflow-x-auto">
              <button
                onClick={() => setActiveTab('marketplace')}
                className={`flex items-center gap-2 px-3.5 py-2 rounded-lg text-sm font-semibold transition whitespace-nowrap ${
                  activeTab === 'marketplace'
                    ? 'bg-emerald-500 text-slate-950 shadow'
                    : 'text-slate-300 hover:text-white hover:bg-slate-700/60'
                }`}
              >
                <Search className="w-4 h-4" />
                <span>Buy Vehicles</span>
              </button>

              <button
                onClick={() => setActiveTab('sell')}
                className={`flex items-center gap-2 px-3.5 py-2 rounded-lg text-sm font-semibold transition whitespace-nowrap ${
                  activeTab === 'sell'
                    ? 'bg-emerald-500 text-slate-950 shadow'
                    : 'text-slate-300 hover:text-white hover:bg-slate-700/60'
                }`}
              >
                <PlusCircle className="w-4 h-4" />
                <span>Sell Your Car</span>
              </button>

              <button
                onClick={() => setActiveTab('admin')}
                className={`flex items-center gap-2 px-3.5 py-2 rounded-lg text-sm font-semibold transition whitespace-nowrap ${
                  activeTab === 'admin'
                    ? 'bg-amber-400 text-slate-950 shadow'
                    : 'text-slate-300 hover:text-white hover:bg-slate-700/60'
                }`}
              >
                <LayoutDashboard className="w-4 h-4" />
                <span>Broker Admin</span>
                {pendingVehicles.length > 0 && (
                  <span className="bg-red-500 text-white text-[11px] font-bold px-1.5 py-0.2 rounded-full">
                    {pendingVehicles.length}
                  </span>
                )}
              </button>
            </nav>

            {/* Desktop Broker Wallet Quick View */}
            <div 
              onClick={() => setActiveTab('admin')}
              className="hidden md:flex items-center gap-3 bg-slate-800/80 hover:bg-slate-800 border border-slate-700 px-3.5 py-1.5 rounded-xl cursor-pointer transition"

            >

                  )}

                  {checkoutState.paymentMethod === 'card' && (
                    <div className="bg-slate-50 p-4 rounded-xl border border-slate-200 space-y-3">
                      <input
                        type="text"
                        required
                        placeholder="Card Number (4532 •••• •••• ••••)"
                        value={checkoutState.cardNumber}
                        onChange={(e) => setCheckoutState({ ...checkoutState, cardNumber: e.target.value })}
                        className="w-full px-3 py-2 bg-white border border-slate-300 rounded-lg text-sm"
                      />
                      <div className="grid grid-cols-2 gap-3">
                        <input
                          type="text"
                          required
                          placeholder="MM/YY"
                          value={checkoutState.cardExpiry}
                          onChange={(e) => setCheckoutState({ ...checkoutState, cardExpiry: e.target.value })}
                          className="px-3 py-2 bg-white border border-slate-300 rounded-lg text-sm"
                        />
                        <input
                          type="password"
                          required
                          maxLength={4}
                          placeholder="CVV"
                          value={checkoutState.cardCvv}
                          onChange={(e) => setCheckoutState({ ...checkoutState, cardCvv: e.target.value })}
                          className="px-3 py-2 bg-white border border-slate-300 rounded-lg text-sm"
                        />
                      </div>
                    </div>
                  )}
                </div>

                {/* Submit Payment Button */}
                <button
                  type="submit"
                  disabled={checkoutState.isProcessing}
                  className="w-full py-3.5 bg-emerald-600 hover:bg-emerald-700 disabled:bg-slate-400 text-white font-extrabold rounded-xl shadow-lg flex items-center justify-center gap-2 transition"
                >
                  {checkoutState.isProcessing ? (
                    <>
                      <RefreshCw className="w-5 h-5 animate-spin" />
                      <span>Verifying Payment & Locking Escrow...</span>
                    </>
                  ) : (
                    <>
                      <Lock className="w-4 h-4" />
                      <span>
                        Authorize {formatZMW(
                          checkoutState.paymentOption === 'deposit'
                            ? Math.round(checkoutVehicle.sellerPrice * (1 + commissionRate / 100) * 0.10)
                            : Math.round(checkoutVehicle.sellerPrice * (1 + commissionRate / 100))
                        )} in Escrow
                      </span>
                    </>
                  )}
                </button>
              </form>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
